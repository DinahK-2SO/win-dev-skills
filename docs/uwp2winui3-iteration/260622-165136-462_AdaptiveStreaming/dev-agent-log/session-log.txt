301.         private async Task AppDownloadedKeyRequest(AdaptiveMediaSourceDownloadRequestedEventArgs args)
302.         {
303.             if (ContentSelectorControl.SelectedModel == null)
304.             {
305.                 return;
306.             }
307. 
308.             // For AzureKeyAcquisitionMethod.ApplicationDownloaded we do the following:
309.             //   Call .GetDeferral() to allow asynchronous work to be performed
310.             //   Get the requested network resource using app code
311.             //   Set .Result.InputStream or .Result.Buffer with the response data
312.             //   Complete the deferral to indicate that we are done.
313.             // With this pattern, the app has complete control over any downloaded content.
314. 
315.             // Obtain a deferral so we can perform asynchronous operations.
316.             var deferral = args.GetDeferral();
317.             try
318.             {
319.                 var appHttpClientForKeys = new HttpClient();
320.                 appHttpClientForKeys.DefaultRequestHeaders.Authorization = new HttpCredentialsHeaderValue("Bearer", ContentSelectorControl.SelectedModel.AesToken);
321. 
322.                 HttpRequestMessage request = new HttpRequestMessage(HttpMethod.Get, args.ResourceUri);
323. 
324.                 HttpResponseMessage response = await appHttpClientForKeys.SendRequestAsync(
325.                     request, HttpCompletionOption.ResponseHeadersRead).AsTask(ctsForAppHttpClientForKeys.Token);
326. 
327.                 if (response.IsSuccessStatusCode)
328.                 {
329.                     args.Result.InputStream = await response.Content.ReadAsInputStreamAsync();
330.                     // Alternatively, we could use:
331.                     // args.Result.Buffer = await response.Content.ReadAsBufferAsync();
332.                 }
333.                 else
334.                 {
335.                     // The app code failed. Report this by setting the args.Result.ExtendedStatus.
336.                     // This will ensure that the AdaptiveMediaSource does not attempt to download the resource
337.                     // itself, and instead treats the download as having failed.
338.                     switch (response.StatusCode)
339.                     {
340.                         case HttpStatusCode.Unauthorized:
341.                             // HTTP_E_STATUS_DENIED
342.                             args.Result.ExtendedStatus = 0x80190191;
343.                             break;
344.                         case HttpStatusCode.NotFound:
345.                             // HTTP_E_STATUS_NOT_FOUND
346.                             args.Result.ExtendedStatus = 0x80190194;
347.                             break;
348.                         default:
349.                             // HTTP_E_STATUS_UNEXPECTED
350.                             args.Result.ExtendedStatus = 0x80190001;
351.                             break;
352.                     }
353.                     Log($"Key Download Failed: {response.StatusCode} {args.ResourceUri}");
354.                 }
355.             }
356.             catch (TaskCanceledException)
357.             {
358.                 Log($"Request canceled: {args.ResourceUri}");
359. 
360.             }
361.             catch (Exception e)
362.             {
363.                 Log($"Key Download Failed: {e.Message} {args.ResourceUri}");
364.             }
365.             finally
366.             {
367.                 // Complete the deferral when done.

50.     public class BitrateHelper
51.     {
52.         // Each bitrate is assigned a symbol as follows:
53.         // The bottom quartile gets 1 bar, the second quartile gets 2 bars,
54.         // the third quartile gets 3 bars, and the top quartile gets 4 bars.
55.         private Dictionary<uint, Symbol> symbolAssignment = new Dictionary<uint, Symbol>();
56. 
57.         public BitrateHelper(IEnumerable<uint> availableBitRates)
58.         {
59.             uint[] sortedBitRates = availableBitRates.ToArray();
60.             Array.Sort(sortedBitRates);
61.             int i = 0;
62.             while (i < sortedBitRates.Length * 1 / 4)
63.             {
64.                 symbolAssignment.Add(sortedBitRates[i++], Symbol.OneBar);
65.             }
66.             while (i < sortedBitRates.Length * 2 / 4)
67.             {
68.                 symbolAssignment.Add(sortedBitRates[i++], Symbol.TwoBars);
69.             }
70.             while (i < sortedBitRates.Length * 3 / 4)
71.             {
72.                 symbolAssignment.Add(sortedBitRates[i++], Symbol.ThreeBars);
73.             }
74.             while (i < sortedBitRates.Length * 4 / 4)
75.             {
76.                 symbolAssignment.Add(sortedBitRates[i++], Symbol.FourBars);
77.             }
78.         }
79. 
80.         public Symbol GetBitrateSymbol(uint currentBitrate)
81.         {
82.             Symbol symbol = Symbol.ZeroBars;
83.             if (symbolAssignment.TryGetValue(currentBitrate, out symbol))
84.             {
85.                 return symbol;
86.             }
87.             return Symbol.ZeroBars;
