35.         private void Pointer_Entered(object sender, PointerRoutedEventArgs e)

42.                 return;
43.             }

46.             penDevice = PenDevice.GetFromPointerId(e.Pointer.PointerId);
47. 
48.             // If a PenDevice cannot be retrieved based on the PointerId, it does not support
49.             // advanced pen features, such as tactile feedback
50.             if (penDevice == null)
51.             {

53.                 supportedFeatures.Text = "";
54.                 supportedFeedback.Text = "";
55.                 return;
56.             }
57.

59.             // has a SimpleHapticsController
60.             hapticsController = penDevice.SimpleHapticsController;
61.             if (hapticsController == null)
62.             {
63.                 statusText.Text = "This pen does not provide tactile feedback";
64.                 return;
65.             }
66. 
67.             // Check which tactile feedback features are supported
68.             supportedFeatures.Text = "Supported Haptics Features:\n";

72.             }
73.             if (hapticsController.IsPlayCountSupported)
74.             {
75.                 supportedFeatures.Text += "PlayCount\n";
76.             }
77.             if (hapticsController.IsPlayDurationSupported)
78.             {
79.                 supportedFeatures.Text += "PlayDuration\n";
80.             }
81.             if (hapticsController.IsReplayPauseIntervalSupported)
82.             {
83.                 supportedFeatures.Text += "ReplayPauseInterval\n";
84.             }
85.

87.             supportedFeedback.Text = "Supported Feedback:\n";

92.                 {
93.                     if (entry.Value == waveform)
94.                     {
