24.         public static void SetUpDeviceScenarios(Dictionary<DeviceType, UIElement> scenarios, UIElement scenarioContainer)
25.         {
26.             UIElement supportedScenario = null;
27. 
28.             if (EventHandlerForDevice.Current.IsDeviceConnected)
29.             {
30.                 foreach (KeyValuePair<DeviceType, UIElement> deviceScenario in scenarios)
31.                 {
32.                     // Enable the scenario if it's generic or the device type matches
33.                     if ((deviceScenario.Key == DeviceType.Any)
34.                         || (deviceScenario.Key == Utilities.GetDeviceType(EventHandlerForDevice.Current.Device)))
35.                     {
36.                         // Make the scenario visible in case other devices use the same scenario and collapsed it.
37.                         deviceScenario.Value.Visibility = Visibility.Visible;
38. 
39.                         supportedScenario = deviceScenario.Value;
40.                     }
41.                     else if (deviceScenario.Value != supportedScenario)    // Don't hide the scenario if it is supported by the current device and is shared by other devices
42.                     {
43.                         deviceScenario.Value.Visibility = Visibility.Collapsed;
44.                     }
45.                 }
46.             }
47. 
48.             if (supportedScenario == null)
49.             {
50.                 // Hide the container so that common elements shared across scenarios are also hidden
51.                 scenarioContainer.Visibility = Visibility.Collapsed;
52. 
53.                 NotifyDeviceNotConnected();
54.             }
55.         }
